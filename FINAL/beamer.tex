\section{Background}

	\begin{frame}{Heat Exchangers}
		Model the flow of heat by analysing:
		\begin{itemize}
			\item Fluid dynamics (Navier-Stokes eqns.)
			\item Heat conduction (Heat eqn.)
		\end{itemize}

		
		
		\includegraphics[width=0.45\textwidth]{imgs/heatdiagram.png}
	\end{frame}
	\begin{frame}{Introduction}
		How is it justified to write an entire project about something so simple and well documented? \pause
		\begin{itemize}
			\item Heat exchangers are traditionally evaluated by use of a body-fitted mesh\pause
			\item For complex geometries, this is unfeasible\pause
			\item Warrants the investigation into alternative methods (Brinkman's method)\pause
			\item Much of the theory is based on the paper \cite{art:TPMS} by Jiang~et~al. 
		\end{itemize}
	\end{frame}
	\begin{frame}{Project Goal}
		The goal of this project is not to optimize a final industrial heat exchanger, but to test a modeling approach.
		
		\begin{itemize}
			\item Can complex TPMS heat exchangers be simulated without body-fitted meshes? \pause
			\item Can two independent fluid channels be represented in one computational domain?\pause
			\item Can heat transfer be modeled using one global temperature field?\pause
			\item We first use a simple sinusoidal wall as a proof-of-concept.
		\end{itemize}
	\end{frame}

\section{TPMS Model}
	\begin{frame}{Background}
		\emph{Triply Periodic Minimal Surfaces} (TPMS) incorporates a wide range of surfaces, of which the \emph{gyroid} is the subject of this project. Discovered by NASA scientist Alan Schoen in 1970, it was proven to be embedded by Grosse-Brauckmann and Wohlgemuth in 1996 \cite{enwiki:gyroid}.
		\begin{itemize}
			\item Embedded; divides space into two \emph{disjoint} sub-volumes. \pause
			\item Can be approximated a the relatively simple equation
			\begin{equation*}
				\sin x\cos y+\sin y\cos z+\sin z\cos x=0
			\end{equation*}
		\end{itemize}\pause
		\centering
		\includegraphics[scale=0.2]{imgs/Gyroid_surface_with_Gaussian_curvature}
		\includegraphics[scale=0.1]{imgs/gyroid_exchanger}
	\end{frame}
	
	\begin{frame}{Advantages of TMPS}
	TPMS provides a compelling alternative to traditional fin-based heat exchangers. 
	\begin{itemize}
		\item High surface-to-volume ratio improves heat exchange.\pause
		\item Continous TPMS walls generally provide good structural integrity (high stiffness).\pause
		\item Geometry can be fine-tuned by function-based parameters. \pause
		\item \emph{Disadvantage:} Difficult to simulate performance
	\end{itemize}
	\end{frame}
	\begin{frame}{Modeling TPMS}
		The same method that was used for the simple sinusoidal geometry is applied to the TMPS structure. 
		
		\begin{itemize}
			\item The geometry is represented by continuous functions. \pause
			\item Two Brinkman flow fields are solved, one for each fluid.\pause
			\item A single temperature field is solved over the whole domain.\pause
			\item Material properties are assigned using smooth interpolation functions.
		\end{itemize}
	\end{frame}
	\begin{frame}{Geometry and boundaries}
		A G-type TPMS structure with 8 periods of 5 mm and a wall thickness of 1.5 mm. The hot-side inlet temperature is 323.15 K, and the cold-side inlet temperature is 273.15 K. The inlet velocities are both 0.01 m/s.
		
		\begin{center}
			\includegraphics[width=0.5\textheight]{imgs/youming}
		\end{center}
		
	\end{frame}
	
\begin{frame}{Material Fields}
	Instead of explicitly meshing the fluid-solid interfaces, the domain is divided by a smooth indicator function $\Phi$.
	
	\begin{columns}[T]
		\begin{column}{0.55\textwidth}
			\begin{align*}
				\Phi &\approx 1\quad \text{fluid 1} \\
				\Phi &\approx 0.5\quad \text{solid wall} \\
				\Phi &\approx 0\quad \text{fluid 2}
			\end{align*}

		\end{column}
		\begin{column}{0.42\textwidth}
			\includegraphics[width=0.9\linewidth]{imgs/indicator}
		\end{column}
	\end{columns}
	\begin{itemize}
		\item The gyroid wall separates the two fluid channels.
		\item Inlet and outlet extensions are added to make boundary conditions easier.
		\item The smooth transition avoids discontinuous material properties.
	\end{itemize}
\end{frame}
\begin{frame}{Flow Model}
	Two Brinkman equations are used to model the two fluids separately.
	
	\begin{itemize}
		\item For fluid 1, the fluid-1 region has low resistance.
		\item The wall and fluid-2 region are treated as high-resistance regions.
		\item Vice versa for fluid 2.
	\end{itemize}
	
	This allows the two flow fields to coexist in the same computational domain without explicitly meshing the interface.
\end{frame}
\begin{frame}{Heat Transfer Model}
	The heat transfer problem is solved using one global temperature field; $\mathbf{u}_{T} =\mathbf{u}_{\text{fluid 1}} +  \mathbf{u}_{\text{fluid 2}}$.
	
	\begin{itemize}
		\item Fluid regions: convection and conduction, Pe=800
		\item Wall region: conduction only, Pe=200
		\item The Péclet number (Pe) is interpolated between fluid and wall regions.
	\end{itemize}
	\begin{center}
		\includegraphics[width=0.45\textwidth]{imgs/peclet}
	\end{center}
\end{frame}

\section{Results}
\begin{frame}{TPMS Simulation}
	The model was solved for a \qty{50}{K} inlet temperature difference.
	
	\begin{itemize}
		\item The two velocity fields remain separated by the TPMS wall.
		\item The temperature field changes across the heat-exchanger core.
		\item Heat is transferred through the TPMS wall from the hot fluid to the cold fluid.
	\end{itemize}

\begin{frame}{Conclusion}
	This method has a certain degree of robustness with respect to mesh discretization. Therefore, even if the local $\Phi$ boundary is rough, as long as the overall flow-channel volume fraction, resistance distribution, and heat-transfer area have similar averaged effects, the outlet temperature and pressure drop may still be close.
\end{frame}
\begin{frame}{Discussion}
	The TPMS model demonstrates that the Brinkman-based method can be extended from a simple wall to a complex gyroid structure.
	
	\begin{itemize}
		\item Advantage: no body-fitted mesh is required for the internal TPMS wall.
		\item Advantage: geometry and wall thickness can be changed by parameters.
		\item Limitation: the result depends on mesh resolution and interpolation width.
	\end{itemize}
\end{frame}
